# Experiment 10 - Data Visualization III

# 1. Import required libraries
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# 2. Load Iris dataset
df = sns.load_dataset('iris')

# 3. Display dataset
print("Iris Dataset:")
print(df.head())

# 4. Check missing values
print("\nMissing Values:")
print(df.isnull().sum())

# 5. Dataset information
print("\nDataset Info:")
print(df.info())

# 6. Statistical summary
print("\nStatistical Summary:")
print(df.describe())

# 7. List features and their types
print("\nFeatures and Data Types:")
for column in df.columns:
    if df[column].dtype == 'object':
        print(column, "-> Nominal (Categorical)")
    else:
        print(column, "-> Numeric")

# ---------------------------------
# Histograms for each feature
# ---------------------------------
numeric_columns = df.select_dtypes(include=['float64', 'int64']).columns

for col in numeric_columns:
    plt.figure(figsize=(6, 4))
    sns.histplot(df[col], kde=True)
    plt.title("Histogram of " + col)
    plt.xlabel(col)
    plt.ylabel("Frequency")
    plt.show()

# ---------------------------------
# Boxplots for each feature
# ---------------------------------
for col in numeric_columns:
    plt.figure(figsize=(6, 4))
    sns.boxplot(y=df[col])
    plt.title("Boxplot of " + col)
    plt.ylabel(col)
    plt.show()

# ---------------------------------
# Comparison and Outlier Detection
# ---------------------------------
print("\nInference:")

print("1. Sepal Length:")
print("- Mostly normally distributed")
print("- Few mild outliers present")

print("\n2. Sepal Width:")
print("- Distribution slightly spread")
print("- Some outliers visible")

print("\n3. Petal Length:")
print("- Clear grouping visible")
print("- Indicates separation between species")

print("\n4. Petal Width:")
print("- Distinct clusters present")
print("- Strong separation among flower species")
print("- Few possible outliers")
