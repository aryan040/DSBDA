# Experiment 1 - Data Wrangling

# 1. Import required libraries
import pandas as pd
import numpy as np

# 2. Load dataset
df = pd.read_csv("data.csv")

# 3. Display dataset
print("First 10 rows:")
print(df.head(10))

print("\nLast 10 rows:")
print(df.tail(10))

# 4. Dataset shape
print("\nDataset Shape:")
print(df.shape)

# 5. Dataset information
print("\nDataset Info:")
print(df.info())

# 6. Statistical summary
print("\nStatistical Summary:")
print(df.describe())

# 7. Check missing values
print("\nMissing Values:")
print(df.isnull().sum())

# 8. Check non-null values
print("\nNon-Null Values:")
print(df.notnull().sum())

# 9. Check data types
print("\nData Types:")
print(df.dtypes)

# 10. Handle missing values
# Fill numerical missing values with mean
df['bore'] = df['bore'].fillna(df['bore'].mean())
df['stroke'] = df['stroke'].fillna(df['stroke'].mean())
df['horsepower'] = df['horsepower'].fillna(df['horsepower'].mean())
df['peak-rpm'] = df['peak-rpm'].fillna(df['peak-rpm'].mean())

# Fill categorical missing values with mode
df['horsepower-binned'] = df['horsepower-binned'].fillna(df['horsepower-binned'].mode()[0])

# 11. Convert data types
df['curb-weight'] = df['curb-weight'].astype(int)

# 12. Normalize numerical columns using Min-Max normalization
numerical_cols = ['horsepower', 'price', 'wheel-base', 'length', 'width', 'height']

for col in numerical_cols:
    df[col] = (df[col] - df[col].min()) / (df[col].max() - df[col].min())

# 13. Encode categorical column
df['aspiration'] = df['aspiration'].map({'std': 0, 'turbo': 1})

# 14. Verify transformed dataset
print("\nUpdated Data Types:")
print(df.dtypes)

print("\nFirst 10 rows after preprocessing:")
print(df.head(10))

# 15. Save cleaned dataset
df.to_csv("cleaned_data.csv", index=False)

print("\nData wrangling completed successfully!")
